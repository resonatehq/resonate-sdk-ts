#!/usr/bin/env python3
"""Tree checker: replays a seam log into a tree of nodes and asserts, at every event, that the transition is legal.
States: ephemeral -> durable -> running -> returned -> settled (local); ephemeral -> durable -> settled (remote).
Durable facts (durable, value known) survive attempts; transient state (running, waits) resets when a task is re-acquired.
Usage: tree.py LOG...   (reads v1 execlogs; the v2 seam log maps 1:1)"""
import json, sys, pathlib, collections

class Node:
    """Knowledge is per task view (worker, task root): a task knows a promise is durable or settled only through its own wire
    responses, preloads and messages, never through another task on the same worker. Transient state is per worker."""
    __slots__ = ("id","parent","scope","ordinal","durable_on","known_on","t","children","minted_in","invoked_ever")
    def __init__(s, id, parent=None, scope=None, ordinal=None):
        s.id, s.parent, s.scope, s.ordinal = id, parent, scope, ordinal
        s.durable_on, s.known_on, s.t, s.children, s.minted_in, s.invoked_ever = set(), set(), {}, [], None, False
    def durable(s, v): return v in s.durable_on
    def known(s, v): return v in s.known_on
    def tr(s, w): return s.t.setdefault(w, {"state": None, "wait_on": set(), "invokes": 0})
    def state(s, w, v=None):                               # the node as seen on worker w (in view v)
        st = s.t.get(w, {}).get("state")
        return st or ("settled" if s.known(v) else "durable" if s.durable(v) else "ephemeral")

class Tree:
    def __init__(s):
        s.nodes = {}; s.viol = []; s.attempt = {}          # (worker, root) -> attempt no
        s.failed = set(); s.decided = set()                 # (worker, root, attempt) sets
        s.open_wire = collections.defaultdict(dict)         # worker -> corr -> (kind, id), fenced sends only
        s.sent = collections.defaultdict(dict)              # worker -> corr -> (kind, id), every send
        s.roots = collections.defaultdict(set)              # worker -> ids that run as their own task there
        s.seen_attempt = {}                                 # (worker, root) -> last attempt id seen, so a change starts a new one
        s.seq = 0
    # ---------------------------------------------------------------- helpers
    def bad(s, rule, node, detail): s.viol.append((rule, s.seq, node, detail))
    def get(s, id, **kw):
        n = s.nodes.get(id)
        if n is None: n = s.nodes[id] = Node(id, **kw)
        return n
    def task_root(s, n, worker):                            # nearest ancestor-or-self that runs as its own task on this worker
        while n.parent is not None and n.id not in s.roots[worker]: n = s.nodes[n.parent]
        return n.id
    def att(s, worker, n): r = s.task_root(n, worker); return (worker, r, s.attempt.get((worker, r), 0))
    def view(s, worker, n): return (worker, s.task_root(n, worker))          # the task whose knowledge applies to n's handling
    def prev_sibling(s, n):
        return next((c for c in s.nodes[n.parent].children if c.ordinal == n.ordinal - 1), None) if n.parent in s.nodes else None
    def reset_attempt(s, worker, root):                     # a new attempt on this worker: its transient state goes, durable facts stay
        for n in s.nodes.values():
            if s.task_root(n, worker) == root: n.t.pop(worker, None)
    # ---------------------------------------------------------------- transitions
    def run_req(s, e):                                      # mint of a child
        n = s.get(e["node"], parent=e["parent"], scope=e["scope"], ordinal=e["ordinal"])
        p = s.get(e["parent"])
        if n not in p.children: p.children.append(n)
        w = e["worker"]; n.minted_in = s.att(w, p); v = s.view(w, p)
        if not n.durable(v) and e["ordinal"] > 1 and s.prev_sibling(n) is None:
            s.bad("ordered_creation", n.id, "ordinal not contiguous: no sibling %d" % (e["ordinal"] - 1))
        if p.state(w) != "running": s.bad("child_minted_while_parent_not_running", n.id, f"parent {p.id} is {p.state(w)}")
    def create_req(s, e, corr):
        n = s.nodes.get(e["id"])
        if n is None: return                                # root create (task.create) or an external: not a child of this tree
        k = s.prev_sibling(n)
        w = e["worker"]; v = s.view(w, s.nodes[n.parent])
